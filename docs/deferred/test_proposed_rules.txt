@pytest.mark.skip(reason='Extension aux commandes sans template reportée à la demande de l’utilisateur.')
@pytest.mark.parametrize('command,context,rule,values',[
 ('no cdp enable','interface E1\n ','if_cdp_off',{'enabled':False}),
 ('no mop enabled','interface E1\n ','if_mop_off',{'feature':'enabled','enabled':False}),
 ('service instance 10 ethernet','interface E1\n ','if_service_instance',{'instance':10}),
 ('bridge-domain 10','interface E1\n service instance 10 ethernet\n  ','if_bridge_domain',{'domain':10}),
 ('encapsulation untagged','interface E1\n service instance 10 ethernet\n  ','if_encap_untagged',{'kind':'untagged'}),
 ('mpls ldp router-id Loopback0','','advanced_mpls_ldp_no_option',{'interface':'Loopback0'}),
 ('service timestamps log datetime msec','','services_timestamp',{'category':'log','format':'datetime','options':'msec'}),
 ('service timestamps debug uptime','','services_timestamp_plain',{'category':'debug','format':'uptime'}),
 ('logging synchronous','line vty 0 4\n ','services_line_synchronous',{'enabled':True}),
 ('privilege level 15','line vty 0 4\n ','services_line_privilege',{'level':15}),
 ('stopbits 1','line con 0\n ','services_line_stopbits',{'bits':1}),
 ('length 0','line vty 0 4\n ','services_line_length',{'length':0}),
 ('transport preferred none','line vty 0 4\n ','services_line_preferred',{'transports':'none'}),
 ('no logging console','','services_console_off',{'enabled':False}),
 ('no ip source-route','','services_source_route_off',{'enabled':False}),
 ('no ip domain lookup','','services_lookup_off',{'enabled':False}),
 ('no service password-encryption','','services_password_encryption_off',{'enabled':False}),
 ('ipv6 unicast-routing','','services_ipv6_routing',{'enabled':True}),
 ('ip http authentication local','','services_http_auth',{'method':'local'}),
 ('ip community-list expanded FILTER deny _65001:','','services_community_expanded',{'name':'FILTER','action':'deny','expression':'_65001:'}),
 ('ip flow-export source Loopback0','','services_flow_source',{'interface':'Loopback0'}),
 ('ip flow-export version 5 origin-as','','services_flow_version',{'version':5,'as_mode':'origin-as'}),
 ('ip flow-cache timeout active 5','','services_flow_timeout',{'type':'active','timeout':5}),
 ('bgp always-compare-med','router bgp 65000\n ','bgp_compare_med',{'enabled':True}),
 ('bgp deterministic-med','router bgp 65000\n ','bgp_deterministic_med',{'enabled':True}),
 ('no bgp default ipv4-unicast','router bgp 65000\n ','bgp_no_default_ipv4',{'enabled':False}),
 ('no synchronization','router bgp 65000\n ','bgp_no_sync',{'enabled':False}),
 ('no auto-summary','router bgp 65000\n ','bgp_no_auto_summary',{'enabled':False}),
 ('no bgp enforce-first-as','router bgp 65000\n ','bgp_no_enforce_first_as',{'enabled':False}),
 ('neighbor 192.0.2.1 advertise additional-paths all','router bgp 65000\n ','bgp_advertise_additional',{'peer':'192.0.2.1','selection':'all'}),
])
def test_new_rule_positive_negative_and_anchoring(command,context,rule,values):
    text=context+command+'\n';line=text.count('\n')
    x=extract(text,official=False)
    assert x['records'][line]['rule']==rule and x['records'][line]['values']==values
    bad=extract(context+'UNSUPPORTED_PREFIX '+command+'\n',official=False)
    assert line not in bad['records']
